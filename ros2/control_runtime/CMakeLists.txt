cmake_minimum_required(VERSION 3.16)
project(control_runtime)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(rclcpp REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/ControlState.msg" "msg/ControlCommand.msg"
  DEPENDENCIES builtin_interfaces)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../native ${CMAKE_CURRENT_BINARY_DIR}/native)
add_executable(control_node src/control_node.cpp)
target_link_libraries(control_node control_native)
ament_target_dependencies(control_node rclcpp)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(control_node "${typesupport}")
install(TARGETS control_node DESTINATION lib/${PROJECT_NAME})
install(TARGETS control_native LIBRARY DESTINATION lib)
install(PROGRAMS scripts/bridge.py DESTINATION lib/${PROJECT_NAME} RENAME bridge)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
