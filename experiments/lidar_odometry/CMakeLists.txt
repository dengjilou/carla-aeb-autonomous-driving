cmake_minimum_required(VERSION 3.16)
project(kitti_odometry LANGUAGES C CXX)
find_package(PCL 1.12 REQUIRED COMPONENTS filters registration)
add_executable(odometry odometry.cpp)
target_compile_features(odometry PRIVATE cxx_std_17)
target_include_directories(odometry PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(odometry PRIVATE ${PCL_LIBRARIES})
enable_testing()
add_test(NAME transform_direction COMMAND odometry --self-test)
