cmake_minimum_required(VERSION 3.16)
project(control_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Eigen3 3.4 REQUIRED NO_MODULE)
add_library(control_native SHARED control_core.cpp)
target_link_libraries(control_native PRIVATE Eigen3::Eigen)
set_target_properties(control_native PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_include_directories(control_native PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(control_native PRIVATE /W4)
else()
  target_compile_options(control_native PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(control_smoke control_smoke.cpp)
target_link_libraries(control_smoke PRIVATE control_native)
add_test(NAME control_smoke COMMAND control_smoke)
